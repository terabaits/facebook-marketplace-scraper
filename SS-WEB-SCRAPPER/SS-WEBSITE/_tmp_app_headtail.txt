31: # Monkey-patch RealDictCursor.execute to work around a psycopg2 bug where
32: # some multiline formatted SQL strings cause "tuple index out of range".
33: # This normalizes the query text only when the original execute raises IndexError.
34: _orig_real_dict_cursor_execute = RealDictCursor.execute
35: def _safe_execute(self, query, vars=None):
36:     try:
37:         return _orig_real_dict_cursor_execute(self, query, vars)
38:     except IndexError:
39:         normalized = query.replace('\r\n', '\n').replace('\n', ' ').replace('  ', ' ')
40:         return _orig_real_dict_cursor_execute(self, normalized, vars)
41: RealDictCursor.execute = _safe_execute
