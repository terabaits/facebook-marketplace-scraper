function renderTableRow(item) {
    // Fix: Don't duplicate producer name if already in cpu_name
    let cpuName;
    if (item.cpu_name) {
        const producer = item.producer || item.vendor || '';
        if (producer && item.cpu_name.toLowerCase().startsWith(producer.toLowerCase())) {
            cpuName = item.cpu_name;
        } else if (producer) {
            cpuName = `${producer} ${item.cpu_name}`;
        } else {
            cpuName = item.cpu_name;
        }
    } else {
        cpuName = item.title || 'Unknown CPU';
    }
    const cores = item.cores ? `${item.cores}c/${item.threads || '?'}t` : 'N/A';
    const modelId = item.matched_cpu_id;

    let priceClass = 'price';
    let priceIndicator = '';

    if (item.price_stats) {
        if (item.price_stats.below_avg) {
            priceClass = 'price-below-avg';
            const savingsPct = Math.round(((item.price_stats.avg - item.price_eur) / item.price_stats.avg) * 100);
            priceIndicator = `<span style="font-size: 0.75rem; color: #27ae60; margin-left: 0.25rem; white-space: nowrap;">↓ -${savingsPct}%</span>`;
        } else {
            priceClass = 'price-above-avg';
            const abovePct = Math.round(((item.price_eur - item.price_stats.avg) / item.price_stats.avg) * 100);
            priceIndicator = `<span style="font-size: 0.75rem; color: #e74c3c; margin-left: 0.25rem; white-space: nowrap;">↑ +${abovePct}%</span>`;
        }
    }

    const confidence = item.confidence_score || 0;
    let confidenceClass = 'confidence-high';
    if (confidence < 0.70) confidenceClass = 'confidence-medium';
    if (confidence < 0.50) confidenceClass = 'confidence-low';

    const rowClass = modelId ? 'clickable' : '';
    const clickHandler = modelId ? `onclick="showModelHistory(${modelId}, '${item.producer} ${item.cpu_name}')"` : '';

    const hasHistory = item.history && item.history.length > 0;

    const cells = columnOrder.map(col => {
        const isColHidden = !columnVisibility[col.id];
        const style = isColHidden ? 'display: none;' : '';
        const hiddenClass = isColHidden ? ' hidden-column' : '';

        switch (col.id) {
            case 'col-image': {
                const rowLocalImage = item.local_image_path ? String(item.local_image_path).replace(/\\/g, '/') : null;
                const rowImageUrl = rowLocalImage ? `/images/${rowLocalImage}` : (item.image_url || null);
                return `<td data-col="col-image" class="${hiddenClass}" style="padding: 0.75rem; text-align: center; ${style}">
                    ${rowImageUrl ? `<img src="${rowImageUrl}" alt="${cpuName}" class="listing-thumb" onclick="event.stopPropagation(); showImageModal('${rowImageUrl}', '${cpuName}')" style="max-width: 80px; max-height: 60px; border-radius: 4px; cursor: pointer;">` : '<div class="listing-thumb-placeholder" style="width: 80px; height: 60px; background: #f0f0f0; border-radius: 4px; display: flex; align-items: center; justify-content: center; color: #999;">📷</div>'}
                </td>`;
            }
            case 'col-cpu':
                // Source dot indicator (small colored dot) - check if enabled in settings
                const showSourceDotsCPU = localStorage.getItem('showSourceDots') !== 'false';
                const sourceDotCPU = showSourceDotsCPU ? (item.source === 'andelemandele'
                    ? `<span class="source-dot" title="Andele Mandele" style="display: inline-block; width: 10px; height: 10px; background: #ff6b6b; border-radius: 50%; margin-left: 8px; cursor: pointer;"></span>`
                    : `<span class="source-dot" title="SS.com" style="display: inline-block; width: 10px; height: 10px; background: #90EE90; border-radius: 50%; margin-left: 8px; cursor: pointer;"></span>`)
                    : '';
                return `<td data-col="col-cpu" class="${hiddenClass}" style="padding: 0.75rem; ${style}">
                    <div style="font-weight: bold;">${cpuName}${sourceDotCPU}</div>
                    ${item.match_method ? `<div style="font-size: 0.85rem; margin-top: 0.25rem;"><span class="${confidenceClass}">${(confidence*100).toFixed(0)}% confidence</span></div>` : ''}
                </td>`;
            case 'col-cores':
                return `<td data-col="col-cores" class="${hiddenClass}" style="padding: 0.75rem; text-align: center; ${style}">${cores}</td>`;
            case 'col-price':
                const hasPriceChange = item.history && item.history.length >= 2 && item.price_eur !== item.history[0].price_eur;
                return `<td data-col="col-price" class="${hiddenClass}" style="text-align: center; ${style}">
                    ${renderPriceDecreaseBadge(item)}
                    <div style="display: flex; align-items: center; justify-content: center; flex-wrap: nowrap;">
                        <span class="${priceClass}">€${item.price_eur}</span>
                        ${priceIndicator}
                    </div>
                    ${showAvgPriceGlobal && item.price_stats ? `<div style="font-size: 0.75rem; color: #888; margin-top: 2px;">avg: €${item.price_stats.avg}</div>` : ''}
                    ${hasPriceChange ? `<br><small style="color: ${item.price_eur < item.history[0].price_eur ? '#e74c3c' : '#27ae60'};">Was: €${item.history[0].price_eur} ${item.price_eur < item.history[0].price_eur ? '↓' : '↑'}</small>` : ''}
                </td>`;
            case 'col-price-perf':
                return `<td data-col="col-price-perf" class="${hiddenClass}" style="font-size: 0.85rem; ${style}">
                    ${item.price_per_r23 ? `€${item.price_per_r23}/R23k<br>` : ''}
                    ${item.price_per_passmark ? `€${item.price_per_passmark}/PMk` : '-'}
                </td>`;
            case 'col-benchmarks':
                return `<td data-col="col-benchmarks" class="${hiddenClass}" style="font-size: 0.85rem; ${style}">
                    ${item.benchmarks ? `
                        ${item.benchmarks.r23_multi ? `R23: ${item.benchmarks.r23_multi.toLocaleString()}<br>` : ''}
                        ${item.benchmarks.r26_multi ? `R26: ${item.benchmarks.r26_multi.toLocaleString()}<br>` : ''}
                        ${item.benchmarks.passmark ? `PM: ${item.benchmarks.passmark.toLocaleString()}` : ''}
                    ` : '-'}
                </td>`;
            case 'col-market':
                return `<td data-col="col-market" class="${hiddenClass}" ${style}>
                    ${item.is_unicorn ? `
                        <span style="background: linear-gradient(135deg, #9333ea, #c084fc); color: white; font-size: 0.75rem; padding: 4px 8px; border-radius: 12px; font-weight: bold; display: inline-block;" title="Only one listing of this model in all history!">🦄 UNICORN</span>
                    ` : (item.price_stats ? `
                        <div style="font-size: 0.875rem;">
                            Avg: €${item.price_stats.avg}<br>
                            Range: €${item.price_stats.min} - €${item.price_stats.max}
                        </div>
                        <div class="price-bar" title="Price percentile: ${item.price_stats.percentile}%">
                            <div class="price-bar-fill" style="width: ${item.price_stats.percentile}%"></div>
                        </div>
                    ` : '<small>No price data</small>')}
                </td>`;
            case 'col-location':
                return `<td data-col="col-location" class="${hiddenClass}" ${style}>${item.seller_location || 'N/A'}</td>`;
            case 'col-date':
                return `<td data-col="col-date" class="${hiddenClass}" ${style}>${item.date_posted ? new Date(item.date_posted).toLocaleDateString() : 'N/A'}</td>`;
            case 'col-actions': {
                // Keep a minimal gear placeholder in every Actions body cell so the column remains
                // reachable when the Actions header is collapsed, but hide the action buttons.
                const isActionsHidden = !columnVisibility['col-actions'];
                const cellPaddingStyle = isActionsHidden ? 'padding: 0.5rem;' : '';
                const actionCellClass = isActionsHidden ? ' actions-collapsed' : '';
                return `<td data-col="col-actions" class="${actionCellClass}" style="${style}${cellPaddingStyle}">
                    ${isActionsHidden
                        ? `<button class="column-settings-btn" onclick="event.stopPropagation(); toggleColumnSettings(event)" title="Column Settings" style="margin:0 auto;">⚙️</button>`
                        : `<div style="display: flex; gap: 0.5rem; flex-wrap: wrap; justify-content: center;">
                            <button class="btn btn-small" onclick="event.stopPropagation(); showListingHistory('${item.listing_id}')">Details</button>
                            <button class="btn btn-small" style="background: #dc3545; color: white; border: none;" onclick="event.stopPropagation(); showFlagModalWithCategories('${item.listing_id ? item.listing_id.replace(/'/g, "\\'").replace(/"/g, '&quot;') : ''}', '${escapeHtml(item.title || '').replace(/'/g, "\\'").replace(/"/g, '&quot;')}')">🚩 Flag</button>
                            <a href="${item.listing_url}" target="_blank" class="btn btn-small" onclick="event.stopPropagation()">View →</a>
                        </div>`}
                </td>`;
            }
            default:
                return '';
        }
    }).join('');

    return `<tr id="listing-row-${item.listing_id}" class="${rowClass}" ${clickHandler} title="${modelId ? 'Click to view model history' : ''}" style="border-bottom: 1px solid #eee;">${cells}</tr>`;
}
